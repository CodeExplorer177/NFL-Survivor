async function submitPick() { const player = playerSelect.value; const team = teamSelect.value; if (!player) { showStatus( "Please select your name.", "error" ); return; } if (!team) { showStatus( "Please select a team.", "error" ); return; } submitButton.disabled = true; submitButton.textContent = "Saving..."; try { // Add a unique timestamp so the browser // cannot use a cached request. const cacheBuster = Date.now(); const url = API_URL + "?action=submitPick" + "&player=" + encodeURIComponent(player) + "&week=" + encodeURIComponent(CURRENT_WEEK) + "&team=" + encodeURIComponent(team) + "&_=" + cacheBuster; const response = await fetch(url, { method: "GET", cache: "no-store" }); const data = await response.json(); if (!data.success) { throw new Error( data.error || "Unable to save pick." ); } // ========================================== // UPDATE FORM IMMEDIATELY // ========================================== currentPick.textContent = "Current Week " + CURRENT_WEEK + " Pick: " + data.team; teamSelect.value = data.team; // ========================================== // SUCCESS MESSAGE // ========================================== if (data.overwritten) { showStatus( "✅ Pick changed from " + data.previousTeam + " to " + data.team + ".", "success" ); } else { showStatus( "✅ Your Week " + CURRENT_WEEK + " pick is " + data.team + ".", "success" ); } // ========================================== // REFRESH GOOGLE SHEET // ========================================== setTimeout( refreshSpreadsheet, 1500 ); // ========================================== // RELOAD PLAYER DATA // ========================================== setTimeout( async function() { await loadPlayerData(player); submitButton.disabled = false; submitButton.textContent = "Change My Pick"; }, 800 ); } catch (error) { showStatus( "❌ " + error.message, "error" ); submitButton.disabled = false; submitButton.textContent = "Change My Pick"; } }
