const SHEET_NAME = "Team Picks"; const HEADER_ROW = 6; const FIRST_PLAYER_ROW = 7; const NUMBER_OF_WEEKS = 18; // ================================================== // MAIN GET REQUEST // ================================================== function doGet(e) { const action = e.parameter.action; const callback = e.parameter.callback; let result; try { if (action === "getPlayers") { result = getPlayers(); } else if (action === "getPlayerData") { result = getPlayerData( e.parameter.player ); } else if (action === "getAllData") { result = getAllData(); } else if (action === "submitPick") { result = submitPick( e.parameter.player, e.parameter.week, e.parameter.team ); } else { result = { success: false, error: "Invalid action." }; } } catch (error) { result = { success: false, error: error.toString() }; } const json = JSON.stringify(result); // ================================================== // JSONP RESPONSE // ================================================== if (callback) { return ContentService .createTextOutput( callback + "(" + json + ");" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } // ================================================== // NORMAL JSON RESPONSE // ================================================== return ContentService .createTextOutput(json) .setMimeType( ContentService.MimeType.JSON ); } // ================================================== // SUBMIT PICK // ================================================== function submitPick(player, week, team) { try { // ------------------------------------------ // Validate inputs // ------------------------------------------ if (!player) { return { success: false, error: "Player is required." }; } if (!team) { return { success: false, error: "Team is required." }; } week = Number(week); if ( week < 1 || week > NUMBER_OF_WEEKS ) { return { success: false, error: "Invalid NFL week." }; } // ------------------------------------------ // Get sheet // ------------------------------------------ const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); if (!sheet) { return { success: false, error: "Sheet not found: " + SHEET_NAME }; } // ------------------------------------------ // Get week headers // ------------------------------------------ const headers = sheet .getRange( HEADER_ROW, 2, 1, NUMBER_OF_WEEKS ) .getValues()[0]; const weekHeader = "Week " + week; const weekIndex = headers.findIndex( header => String(header).trim() === weekHeader ); if (weekIndex === -1) { return { success: false, error: "Week column not found: " + weekHeader }; } const weekColumn = 2 + weekIndex; // ------------------------------------------ // Find player // ------------------------------------------ const lastRow = sheet.getLastRow(); const playerValues = sheet .getRange( FIRST_PLAYER_ROW, 1, lastRow - FIRST_PLAYER_ROW + 1, 1 ) .getValues(); let playerRow = -1; for ( let i = 0; i < playerValues.length; i++ ) { const name = String( playerValues[i][0] ).trim(); if ( name && name.toLowerCase() === String(player) .trim() .toLowerCase() ) { playerRow = FIRST_PLAYER_ROW + i; break; } } if (playerRow === -1) { return { success: false, error: "Player not found." }; } // ------------------------------------------ // Actual picks are one row below player // ------------------------------------------ const pickRow = playerRow + 1; // ------------------------------------------ // Get existing picks // ------------------------------------------ const playerPicks = sheet .getRange( pickRow, 2, 1, NUMBER_OF_WEEKS ) .getValues()[0]; // ------------------------------------------ // Existing pick for current week // ------------------------------------------ const oldPick = String( playerPicks[weekIndex] || "" ).trim(); // ------------------------------------------ // Check if team was already used // ------------------------------------------ for ( let i = 0; i < playerPicks.length; i++ ) { if (i === weekIndex) { continue; } const previousPick = String( playerPicks[i] || "" ).trim(); if ( previousPick && previousPick.toLowerCase() === String(team) .trim() .toLowerCase() ) { return { success: false, error: "You have already used the " + team + "." }; } } // ------------------------------------------ // Write pick // ------------------------------------------ sheet .getRange( pickRow, weekColumn ) .setValue(team); SpreadsheetApp.flush(); // ------------------------------------------ // Return success // ------------------------------------------ return { success: true, player: player, week: week, team: team, previousTeam: oldPick || null, overwritten: oldPick !== "" }; } catch (error) { return { success: false, error: error.toString() }; } } // ================================================== // GET PLAYERS // ================================================== function getPlayers() { try { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); if (!sheet) { return { success: false, error: "Sheet not found: " + SHEET_NAME }; } const lastRow = sheet.getLastRow(); const values = sheet .getRange( FIRST_PLAYER_ROW, 1, lastRow - FIRST_PLAYER_ROW + 1, 1 ) .getValues(); const players = []; for ( let i = 0; i < values.length; i++ ) { const name = String( values[i][0] ).trim(); if (name) { players.push(name); } } return { success: true, players: players }; } catch (error) { return { success: false, error: error.toString() }; } } // ================================================== // GET PLAYER DATA // ================================================== function getPlayerData(player) { try { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); if (!sheet) { return { success: false, error: "Sheet not found: " + SHEET_NAME }; } const headers = sheet .getRange( HEADER_ROW, 2, 1, NUMBER_OF_WEEKS ) .getValues()[0]; const lastRow = sheet.getLastRow(); const players = sheet .getRange( FIRST_PLAYER_ROW, 1, lastRow - FIRST_PLAYER_ROW + 1, 1 ) .getValues(); for ( let i = 0; i < players.length; i++ ) { const name = String( players[i][0] ).trim(); if ( name && name.toLowerCase() === String(player) .trim() .toLowerCase() ) { const playerRow = FIRST_PLAYER_ROW + i; const pickRow = playerRow + 1; const picks = sheet .getRange( pickRow, 2, 1, NUMBER_OF_WEEKS ) .getValues()[0]; const result = {}; for ( let j = 0; j < NUMBER_OF_WEEKS; j++ ) { const pick = String( picks[j] || "" ).trim(); if (pick) { result[ String( headers[j] ).trim() ] = pick; } } return { success: true, player: name, picks: result }; } } return { success: false, error: "Player not found." }; } catch (error) { return { success: false, error: error.toString() }; } } // ================================================== // GET ALL SURVIVOR DATA // // This reads the LIVE Google Sheet directly. // It does NOT use the published Google Sheet. // ================================================== function getAllData() { try { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); if (!sheet) { return { success: false, error: "Sheet not found: " + SHEET_NAME }; } // ------------------------------------------ // Get headers // ------------------------------------------ const headers = sheet .getRange( HEADER_ROW, 2, 1, NUMBER_OF_WEEKS ) .getValues()[0] .map( header => String(header).trim() ); // ------------------------------------------ // Get player names // ------------------------------------------ const lastRow = sheet.getLastRow(); const numberOfRows = lastRow - FIRST_PLAYER_ROW + 1; const names = sheet .getRange( FIRST_PLAYER_ROW, 1, numberOfRows, 1 ) .getValues(); // ------------------------------------------ // Get all week data // ------------------------------------------ const allValues = sheet .getRange( FIRST_PLAYER_ROW, 2, numberOfRows, NUMBER_OF_WEEKS ) .getValues(); const players = []; // ------------------------------------------ // Each player's actual picks are one row // below their name row. // ------------------------------------------ for ( let i = 0; i < names.length; i++ ) { const playerName = String( names[i][0] ).trim(); if (!playerName) { continue; } const pickRowIndex = i + 1; if ( pickRowIndex >= allValues.length ) { continue; } const row = allValues[ pickRowIndex ]; const picks = {}; for ( let week = 0; week < NUMBER_OF_WEEKS; week++ ) { const team = String( row[week] || "" ).trim(); picks[ headers[week] ] = team; } players.push({ name: playerName, picks: picks }); } return { success: true, headers: headers, players: players }; } catch (error) { return { success: false, error: error.toString() }; } }
